// Firebase Storage Security Rules for Sriox Careers Dashboard
// Copy these rules to your Firebase Console > Storage > Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // Resume uploads - allow anyone to upload, admin to read/delete
    match /resumes/{applicationId}/{fileName} {
      // Allow anyone to upload resumes (job applicants)
      allow create: if request.auth == null || request.auth != null
        && isValidResumeFile(request.resource)
        && request.resource.size < 10 * 1024 * 1024; // 10MB limit
      
      // Allow admin to read and delete resumes
      allow read, delete: if request.auth != null 
        && request.auth.token.email == 'sridatta.k99@gmail.com';
      
      // Allow applicant to read their own resume for 1 hour after upload
      allow read: if request.time < resource.timeCreated + duration.value(1, 'h');
    }
    
    // Admin uploads (company logos, documents, etc.)
    match /admin/{fileName} {
      allow read, write, delete: if request.auth != null 
        && request.auth.token.email == 'sridatta.k99@gmail.com'
        && request.resource.size < 50 * 1024 * 1024; // 50MB limit
    }
    
    // Public assets (logos, images for job postings)
    match /public/{fileName} {
      // Allow read access to everyone
      allow read: if true;
      
      // Allow admin to upload/modify public assets
      allow write, delete: if request.auth != null 
        && request.auth.token.email == 'sridatta.k99@gmail.com'
        && isValidImageFile(request.resource)
        && request.resource.size < 20 * 1024 * 1024; // 20MB limit
    }
    
    // Job posting attachments
    match /job_attachments/{jobId}/{fileName} {
      // Allow public read access for job-related files
      allow read: if true;
      
      // Allow admin to upload job attachments
      allow write, delete: if request.auth != null 
        && request.auth.token.email == 'sridatta.k99@gmail.com'
        && request.resource.size < 25 * 1024 * 1024; // 25MB limit
    }
    
    // Helper functions for file validation
    function isValidResumeFile(file) {
      return file.contentType.matches('application/pdf') ||
             file.contentType.matches('application/msword') ||
             file.contentType.matches('application/vnd.openxmlformats-officedocument.wordprocessingml.document') ||
             file.contentType.matches('text/plain');
    }
    
    function isValidImageFile(file) {
      return file.contentType.matches('image/.*');
    }
  }
}
